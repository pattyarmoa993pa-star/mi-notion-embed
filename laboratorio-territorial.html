Un año de simulación, en pseudocódigo 
def tick(S):
    S.t += 1
    ejecutar_cola(S)            # acciones institucionales que vencen
    for n in S.nodos:
        n.poblacion *= 1.008
        n.extranjeros += n.extranjeros * tasa * (1 + 0.1 * n.peso_intl)
        for _ in poisson(lambda_relaciones(n)):
            crear_o_reforzar_arista(n)       # intl 55% / conocimiento 45%
    aplicar_eventos(S)          # +20% población, ruta cortada...
    institucion(S)              # registra las R aristas con mayor brecha (retardo d)
    capacidad *= 1.008          # el Estado crece más lento que la sociedad
    metricas(S)                 # presión, componentes, brecha
Fórmulas clave 
presion_i   = (poblacion_i + derivaciones_hijos + desborde_hijos) / capacidad_i
brecha_red  = 1 - Σ min(peso_real, peso_registrado) / Σ peso_real
brecha_datos= Σ |N_real - N_registrado| / Σ N_real
BRECHA      = 0.6·brecha_red + 0.4·min(1, 4·brecha_datos)
